# V2-2c Vision UX Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Surface the V2-2b vision pipeline to users — a clinician camera-link health panel, a patient framing-check step, rendering of the `kinematic_score`/`compensation_flag` frame fields, and console fusion controls.

**Architecture:** A new sidecar `VisionHealthMonitor` samples WebRTC connection state, frame rate, and kinematic freshness, broadcast on a new clinician-only `vision_health` WS frame (mirrors the `eeg_quality` pattern, ~2 Hz). The console parses it into a store and renders a health panel + kinematic readout, and gains fusion controls (mode A/B/C, affected side, calibrate rest pose) wired through the existing IPC → control-socket path. The patient app gains a browser-local framing-check scene (lightweight luminance heuristic + manual confirm) and a kinematic overlay (compensation cue + verified indicator) driven by the score frame it already receives.

**Tech Stack:** Python 3.10 (`eeg` conda env, `aiortc`, `websockets`, `pytest`); TypeScript + React + Vitest (clinician-console Electron, patient-web Vite).

## Global Constraints

- All Python runs via `conda run -n eeg`; run `pytest` from the repo root so `signal-engine/tests/conftest.py` is discovered. Test command: `conda run -n eeg python -m pytest signal-engine/tests/ -m "not live"`.

- `mediapipe` stays pinned at `0.10.14` (do not upgrade).

```python
"""Samples camera-link health for the clinician `vision_health` WS frame.

Pure sampler: no threads, no asyncio. The WS server calls `sample()` on a fixed
cadence and serializes the dict. `fps` is derived from `VideoFrameSink.frame_count`
deltas between calls; `kinematic_active` reflects whether a fresh KinematicResult
is present; `degraded` is only raised in kinematic-weighted modes (B/C) — Mode A
is EEG-only by design and never "degraded".
"""
from __future__ import annotations

import time


class VisionHealthMonitor:
    def __init__(self, receiver, sink, latest_kinematic, engine, *,
                 staleness_s: float = 0.3, now=time.time):
        self.receiver = receiver
        self.sink = sink
        self.latest = latest_kinematic
        self.engine = engine
        self.staleness_s = float(staleness_s)
        self._now = now
        self._prev_count = None
        self._prev_t = None

    def sample(self) -> dict:
        t = self._now()
        count = self.sink.frame_count
        fps = 0.0
        if self._prev_count is not None and self._prev_t is not None:
            dt = t - self._prev_t
            if dt > 0:
                fps = (count - self._prev_count) / dt
        self._prev_count = count
        self._prev_t = t

        result = self.latest.get()
        kinematic_active = (
            result is not None and (t - result.timestamp) <= self.staleness_s
        )
        degraded = self.engine.session_mode in ("B", "C") and not kinematic_active

        return {
            "connection": self.receiver.connection_state(),
            "fps": round(float(fps), 1),
            "kinematic_active": bool(kinematic_active),
            "degraded": bool(degraded),
        }
```

```python
            pass
```

```ts
// Parser for the sidecar's clinician-only `vision_health` frame (V2-2c, ~2 Hz).
// Defensive like the score/quality parsers: never throw on a bad frame.

export type VisionHealthFrame = {
  connection: string;
  fps: number;
  kinematicActive: boolean;
  degraded: boolean;
  timestamp: number;
};

export type VisionHealthParseResult =
  | { ok: true; frame: VisionHealthFrame }
  | { ok: false; error: string };

export function parseVisionHealthMessage(raw: string): VisionHealthParseResult {
  let data: unknown;
  try {
    data = JSON.parse(raw);
  } catch {
    return { ok: false, error: "malformed JSON" };
  }
  if (typeof data !== "object" || data === null) {
    return { ok: false, error: "not an object" };
  }
  const d = data as Record<string, unknown>;
  if (d.type !== "vision_health") {
    return { ok: false, error: `unexpected type: ${String(d.type)}` };
  }
  return {
    ok: true,
    frame: {
      connection: typeof d.connection === "string" ? d.connection : "closed",
      fps: typeof d.fps === "number" && Number.isFinite(d.fps) ? d.fps : 0,
      kinematicActive: Boolean(d.kinematic_active),
      degraded: Boolean(d.degraded),
      timestamp: typeof d.timestamp === "number" ? d.timestamp : 0,
    },
  };
}
```

```ts
import type { VisionHealthFrame } from "./visionHealthMessage";

// Latest-only store for vision_health frames. Mirrors EegQualityStore's
// stable-snapshot pattern so it plugs into useSyncExternalStore without looping.

export type VisionHealthState = { latest: VisionHealthFrame | null };

const EMPTY: VisionHealthState = { latest: null };

export class VisionHealthStore {
  private subscribers = new Set<(s: VisionHealthState) => void>();
  private snapshot: VisionHealthState = EMPTY;

  push(frame: VisionHealthFrame): void {
    this.snapshot = { latest: frame };
    for (const cb of this.subscribers) cb(this.snapshot);
  }

  getState(): VisionHealthState {
    return this.snapshot;
  }

  subscribe(cb: (s: VisionHealthState) => void): () => void {
    this.subscribers.add(cb);
    return () => {
      this.subscribers.delete(cb);
    };
  }
}
```

```ts
export function useVisionHealthStore(store: VisionHealthStore): VisionHealthState {

    (cb) => store.subscribe(cb),
    () => store.getState(),
  );
```

```tsx
type Props = {
```

```tsx
  onAffectedSide?: (side: "left" | "right") => void;

export function CalibrateStep({ onNext, onBack, onFusionMode, onAffectedSide, onCalibratePose, defaultSide = "right" }: Props) {

                key={m.id}
                className={`btn ${mode === m.id ? "btn-primary" : ""}`}

              >
                {m.label} · {m.hint}
              </button>
            ))}
          </div>

            <span className="fusion-label">Affected side</span>
```

```tsx
 */
const MIN_LUMA = 40; // 0..255 average; below this the frame is too dark

export function FramingCheckScene({ onConfirm }: { onConfirm: () => void }) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [ready, setReady] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const md = navigator.mediaDevices;
    if (!md?.getUserMedia) {
      setError("Camera unavailable");
      return;
    }
    let stream: MediaStream | null = null;
    let cancelled = false;
    let raf = 0;

    const sample = () => {
      const v = videoRef.current;
      const c = canvasRef.current;
      if (v && c && v.videoWidth > 0 && v.videoHeight > 0) {
        const ctx = c.getContext("2d");
        if (ctx) {
          c.width = 32; c.height = 24;
          ctx.drawImage(v, 0, 0, 32, 24);
          const { data } = ctx.getImageData(0, 0, 32, 24);
          let sum = 0;
          for (let i = 0; i < data.length; i += 4) sum += (data[i] + data[i + 1] + data[i + 2]) / 3;
          const luma = sum / (data.length / 4);
          const live = stream?.getVideoTracks()[0]?.readyState === "live";
          setReady(Boolean(live) && luma >= MIN_LUMA);
        }
      }
      raf = requestAnimationFrame(sample);
    };

    md.getUserMedia({ video: { facingMode: "user" }, audio: false })
      .then((s) => {
        if (cancelled) { s.getTracks().forEach((t) => t.stop()); return; }
        stream = s;
        if (videoRef.current) videoRef.current.srcObject = s;
        raf = requestAnimationFrame(sample);
      })
      .catch(() => setError("Camera unavailable"));

    return () => {
      cancelled = true;
      cancelAnimationFrame(raf);
      stream?.getTracks().forEach((t) => t.stop());
    };
  }, []);

  return (
    <main className="scene scene--framing">
      <h1 className="framing-title">Let's get you in frame</h1>
      <p className="framing-hint">Sit so your head, shoulders, and reaching arm fit inside the box.</p>
      <div className="framing-stage">
        {error ? (
          <div className="framing-placeholder">{error}</div>
        ) : (
          <video ref={videoRef} autoPlay playsInline muted className="framing-video" />
        )}
        <div className="framing-guide" data-testid="framing-guide" aria-hidden="true" />
        <canvas ref={canvasRef} style={{ display: "none" }} />
      </div>
      <div className="framing-actions">
        <button
          type="button"
          className="btn"
          data-testid="framing-skip"
          onClick={onConfirm}
        >
          Skip (no camera)
        </button>
        <button
          type="button"
          className="btn btn-primary"
          data-testid="framing-confirm"
          disabled={!ready}
          onClick={onConfirm}
        >
          I'm in frame ▶
        </button>
      </div>
    </main>
  );
}

export default FramingCheckScene;
```

```tsx
import { useEffect, useRef, useState } from "react";
import type { ScoreStore } from "../scoring/scoreStore";

/**
 * V2-2c patient render of the vision fields on the score frame.
 *  - P4 "movement verified": shown whenever kinematicScore is present (vision
 *    is contributing to the fused score that grows the plant/reach).
 *  - P5 compensation cue: a non-punitive reminder to keep the trunk still,
 *    debounced with hysteresis so it never flickers frame-to-frame.
 * Never blocks play; the penalty is already applied upstream in intent_score.
 */
type Props = {
  store: ScoreStore;
  showAfterMs?: number;
  hideAfterMs?: number;
  pollMs?: number;
};

export function KinematicOverlay({ store, showAfterMs = 400, hideAfterMs = 600, pollMs = 100 }: Props) {
  const [verified, setVerified] = useState(false);
  const [showComp, setShowComp] = useState(false);
  const flaggedSince = useRef<number | null>(null);
  const clearedSince = useRef<number | null>(null);

  useEffect(() => {
    const id = setInterval(() => {
      const latest = store.latest();
      const now = Date.now();
      setVerified(latest != null && latest.kinematicScore !== null);

      const flagged = Boolean(latest?.compensationFlag);
      if (flagged) {
        clearedSince.current = null;
        if (flaggedSince.current === null) flaggedSince.current = now;
        if (now - flaggedSince.current >= showAfterMs) setShowComp(true);
      } else {
        flaggedSince.current = null;
        if (clearedSince.current === null) clearedSince.current = now;
        if (now - clearedSince.current >= hideAfterMs) setShowComp(false);
      }
    }, pollMs);
    return () => clearInterval(id);
  }, [store, showAfterMs, hideAfterMs, pollMs]);

  return (
    <div className="kin-overlay" aria-live="polite">
      {verified && (
        <span className="kin-verified" data-testid="verified-badge">✓ movement verified</span>
      )}
      {showComp && (
        <span className="kin-comp" data-testid="compensation-cue">
          Keep your trunk still — reach with your arm
        </span>
      )}
    </div>
  );
}

export default KinematicOverlay;
```
